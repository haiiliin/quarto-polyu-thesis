% Command for creating the thesis title page in accordance with the

\newcommand{\thesiscoverpage}[1]{
  % cover page
  \ifx\thesisVersion\empty\else
    \begin{titlepage}
      \pdfbookmark[0]{Cover}{Cover}
      \flushright
      \hfill
      \vfill
      {\LARGE\thesisTitle \par}
      \rule[5pt]{\textwidth}{.4pt} \par
      {\Large\thesisName}
      \vfill
      \textit{\large\thesisDate} \\
      Version: \thesisVersion
    \end{titlepage}
  \fi

  % main cover page
  \begin{titlepage}
    \pdfbookmark[0]{Cover Page}{Cover Page}
    % \tgherosfont
    \centering

    \includegraphics[width=6cm]{PolyU_Logo_CMKY} \\[4mm]
    {\Large \thesisUniversity} \\[4mm]
    \textsf{\thesisUniversityDepartment} \\
    \textsf{\thesisUniversityInstitute} \\
    %\textsf{\thesisUniversityGroup} \\

    \vfill
    % {\large \thesisSubject} \\[5mm]
    {\LARGE\bfseries\color{thesismain} \MakeUppercase{\thesisTitle} \par} \vspace{10mm}
    {\Large \MakeUppercase{\thesisName}} \\[5mm]

    \vfill
    {\large \thesisDegree} \\[5mm]
    {\Large \thesisUniversity} \\[5mm]
    \ifx\thesisPartnerUniversity\empty\else
      {\large
      This programme is jointly offered by \thesisUniversity\ and \thesisPartnerUniversity \par} \vspace{5mm}
    \fi

    \vfill
    {\large \thesisDate} \\[5mm]
    \ifthenelse{\equal{$initial$}{true}}{\large Initial Submission for Examination Purpose}{}
  \end{titlepage}
}
